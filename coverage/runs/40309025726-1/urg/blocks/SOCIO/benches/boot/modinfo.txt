261        1/1              request_region = address_region(vs_if.req.addr);
262        1/1              request_cacheable = HAS_CACHE && control_if.control.cfg.cache_enable && request_region != REGION_INFER &&
263                                             (!vs_if.req.write || request_region == REGION_COMB);
264                     
265        1/1              match_found = 1'b0;
266        1/1              match_slot = '0;
267        1/1              set_conflict = 1'b0;
268        1/1              free_found = 1'b0;
269        1/1              free_slot = '0;
270        1/1              for (int unsigned slot = 0; slot < MSHRS; slot++) begin
271        1/1                if (!free_found && mshr_state_r[slot] == MSHR_INACTIVE) begin
272        1/1                  free_found = 1'b1;
273        1/1                  free_slot = slot[MSHR_W-1:0];
274                           end
                        MISSING_ELSE
275        1/1                if (!match_found && mshr_state_r[slot] != MSHR_INACTIVE &&
276                               mshr_word_addr_r[slot] == vs_if.req.addr) begin
277        0/1     ==>          match_found = 1'b1;
278        0/1     ==>          match_slot = slot[MSHR_W-1:0];
279                           end
                        MISSING_ELSE
280        1/1                if (request_cacheable && mshr_state_r[slot] != MSHR_INACTIVE &&
281                               cache_set(mshr_word_addr_r[slot]) == cache_set(vs_if.req.addr))
282        0/1     ==>          set_conflict = 1'b1;
                        MISSING_ELSE
283                         end
284                     
285        1/1              wm_match_found = 1'b0;
286        1/1              wm_match_slot = '0;
287        1/1              wm_free_found = 1'b0;
288        1/1              wm_free_slot = '0;
289        1/1              wm_set_conflict = 1'b0;
290        1/1              for (int unsigned slot = 0; slot < WRITE_MISS_ENTRIES; slot++) begin
291        1/1                if (!wm_free_found && wm_state_r[slot] == WM_INACTIVE) begin
292        1/1                  wm_free_found = 1'b1;
293        1/1                  wm_free_slot = slot[WM_W-1:0];
294                           end
                        MISSING_ELSE
295        1/1                if (!wm_match_found && wm_state_r[slot] != WM_INACTIVE &&
296                               wm_word_addr_r[slot] == vs_if.req.addr) begin
297        0/1     ==>          wm_match_found = 1'b1;
298        0/1     ==>          wm_match_slot = slot[WM_W-1:0];
299                           end
                        MISSING_ELSE
300        1/1                if (request_cacheable && wm_state_r[slot] != WM_INACTIVE &&
301                               cache_set(wm_word_addr_r[slot]) == cache_set(vs_if.req.addr))
302        0/1     ==>          wm_set_conflict = 1'b1;
                        MISSING_ELSE
303                         end
304                     
305                         // A read can join an existing refill only while that MSHR still accepts
306                         // clients. Once it is replaying, the response is already being drained.
307        1/1              request_merge = !vs_if.req.write && match_found &&
308                                         (mshr_state_r[match_slot] == MSHR_REQUESTING ||
309                                          mshr_state_r[match_slot] == MSHR_WRITE_TO_SRAM);
310                         // Same-address writes merge into the sidecar entry. The newest data wins
311                         // only on bits selected by the newest mask.
312        1/1              request_merge_write = request_cacheable && vs_if.req.write && wm_match_found;
313                         // A COMB write can join a pending read refill. It allocates a write-miss
314                         // entry, but deliberately does not allocate another read MSHR or issue a
315                         // duplicate FPGA read.
316        1/1              request_join_read = request_cacheable && vs_if.req.write && !wm_match_found && match_found &&
317                                             (mshr_state_r[match_slot] == MSHR_REQUESTING ||
318                                              mshr_state_r[match_slot] == MSHR_WRITE_TO_SRAM);
319                         // Direct mapping means an in-flight refill for the same cache set must
320                         // finish first. Otherwise two responses could overwrite one cache slot in
321                         // an order that does not match the requests that caused them.
322        1/1              request_lookup = request_cacheable && !set_conflict && !wm_set_conflict;
323        1/1              request_bypass_read = !request_cacheable && !vs_if.req.write && !match_found;
324        1/1              request_direct_write = !request_cacheable && vs_if.req.write;
325                     
326        1/1              request_accept = 1'b0;
327        1/1              if (vs_if.req.valid && !control_if.control.vs_writeback && !response_r.valid && !backing_if.rsp.valid) begin
328        0/1     ==>        if (request_merge)
329        0/1     ==>          request_accept = 1'b1;
330        0/1     ==>        else if (request_merge_write)
331        0/1     ==>          request_accept = 1'b1;
332                           // A pending cache lookup may need this same free write-miss slot on the
333                           // next edge. Do not allocate a joined entry until that lookup resolves.
334        0/1     ==>        else if (!lookup_pending_r && request_join_read && wm_free_found)
335        0/1     ==>          request_accept = 1'b1;
336                           // Reserve dirty-victim space before lookup to avoid blocking the
337                           // backing response that frees a full writeback FIFO.
338        0/1     ==>        else if (!lookup_pending_r && request_lookup && cache_available &&
339                                    wb_count_r < WRITEBACK_DEPTH &&
340                                    (vs_if.req.write ? wm_free_found : free_found))
341        0/1     ==>          request_accept = 1'b1;
342        0/1     ==>        else if (!lookup_pending_r && request_bypass_read && free_found)
343        0/1     ==>          request_accept = 1'b1;
344        0/1     ==>        else if (!lookup_pending_r && request_direct_write && direct_state_r == DIRECT_IDLE &&
345                                    (request_region != REGION_INFER || !match_found))
346        0/1     ==>          request_accept = 1'b1;
                   ==>  MISSING_ELSE
347                         end
                        MISSING_ELSE
348        1/1              vs_if.ack = request_accept;
349                       end
350                     
351                       // A tag match is meaningful only when the cache bank reports the set valid.
352                       // The tag/data result belongs to lookup_req_r, not to the request currently
353                       // visible at the arbiter, which may already be a different request.
354                       assign lookup_hit = cache_lookup_valid &&
355                                           cache_lookup_tag == cache_tag(lookup_req_r.addr);
356                     
357                       always_comb begin
358                         // A dirty victim must be preserved before its set is replaced. A clean
359                         // victim can be overwritten directly, and a hit never needs a victim.
360        1/1              lookup_can_finish = lookup_pending_r &&
361                                             (lookup_hit || !cache_lookup_valid || !cache_lookup_dirty ||
362                                              wb_count_r < WRITEBACK_DEPTH);
363        1/1              lookup_write_value = masked_merge(cache_lookup_data, lookup_req_r.wdata,
364                                                           lookup_req_r.wmask);
365                       end

367                       // Select the next read MSHR that can issue a backing request. The round-robin
368                       // pointer prevents one client from permanently starving other MSHRs, while
369                       // the memory link still serializes the physical request bytes.
370                       always_comb begin
371        1/1              issue_found = 1'b0;
372        1/1              issue_slot = issue_rr_r;
373        1/1              for (int unsigned offset = 0; offset < MSHRS; offset++) begin
374        1/1                if (!issue_found &&
375                               mshr_state_r[(issue_rr_r + offset) % MSHRS] == MSHR_REQUESTING) begin
376        0/1     ==>          issue_found = 1'b1;
377        0/1     ==>          issue_slot = (issue_rr_r + offset) % MSHRS;
378                           end
                        MISSING_ELSE
379                         end
380                       end
381                     
382                       // A non-joined write miss issues the same four-byte backing read as a read
383                       // MSHR. Its stored mask/data are applied only when that response returns.
384                       always_comb begin
385        1/1              wm_issue_found = 1'b0;
386        1/1              wm_issue_slot = wm_issue_rr_r;
387        1/1              for (int unsigned offset = 0; offset < WRITE_MISS_ENTRIES; offset++) begin
388        1/1                if (!wm_issue_found &&
389                               wm_state_r[(wm_issue_rr_r + offset) % WRITE_MISS_ENTRIES] == WM_REQUESTING) begin
390        0/1     ==>          wm_issue_found = 1'b1;
391        0/1     ==>          wm_issue_slot = (wm_issue_rr_r + offset) % WRITE_MISS_ENTRIES;
392                           end
                        MISSING_ELSE
393                         end
394                       end
395                     
396                       // Replay one completed read to all clients that merged onto its word.
397                       // There is no response-ready signal in the tile protocol, so the client bit
398                       // is consumed when the one-cycle response is emitted. The arbiter is
399                       // expected to consume that response and the tile must be ready when it has
400                       // an outstanding accepted read.
401                       always_comb begin
402        1/1              replay_found = 1'b0;
403        1/1              replay_slot = '0;
404        1/1              replay_client = '0;
405        1/1              for (int unsigned slot = 0; slot < MSHRS; slot++) begin
406        1/1                for (int unsigned client = 0; client < VS_CLIENT_IDS; client++) begin
407        1/1                  if (!replay_found && mshr_state_r[slot] == MSHR_REPLAYING &&
408                                 mshr_clients_r[slot][client]) begin
409        0/1     ==>            replay_found = 1'b1;
410        0/1     ==>            replay_slot = slot[MSHR_W-1:0];
411        0/1     ==>            replay_client = client[2:0];
412                             end
                        MISSING_ELSE
413                           end
414                         end
415                       end
416                     
417                       // Response context is local metadata. The FPGA only preserves ordering;
418                       // these values let us route the response to MSHR, direct-write, or WB logic.
419                       // Context values 0..MSHRS-1 identify read MSHRs. The reserved values 16 and
420                       // 17 identify the direct write and writeback paths. Values beginning at 18
421                       // identify parameterized write-miss entries.
422                       assign rsp_is_mshr = backing_if.rsp.response_context < MSHRS;
423                       assign rsp_mshr_slot = backing_if.rsp.response_context[MSHR_W-1:0];
424                       assign rsp_is_write_miss =
425                           backing_if.rsp.response_context >= RESPONSE_CONTEXT_WRITE_MISS_BASE &&
426                           backing_if.rsp.response_context < RESPONSE_CONTEXT_WRITE_MISS_BASE + WRITE_MISS_ENTRIES;
427                       assign rsp_wm_slot = backing_if.rsp.response_context - RESPONSE_CONTEXT_WRITE_MISS_BASE;
428                       assign rsp_is_direct = backing_if.rsp.response_context == RESPONSE_CONTEXT_TILE_WRITE;
429                       assign rsp_is_wb = backing_if.rsp.response_context == RESPONSE_CONTEXT_WRITEBACK;
430                     
431                       always_comb begin
432                         // A response is accepted only when its owner can commit the result. If the
433                         // cache is busy with a synchronous lookup or boundary walk, rsp_ready is
434                         // held low and the memory link keeps the response stable.
435        1/1              rsp_wm_match_found = 1'b0;
436        1/1              rsp_wm_match_slot = '0;
437        1/1              mshr_rsp_ready = 1'b0;
438        1/1              wm_rsp_ready = 1'b0;
439        1/1              if (rsp_is_mshr) begin
440        1/1                mshr_rsp_ready = mshr_state_r[rsp_mshr_slot] == MSHR_WRITE_TO_SRAM &&
441                                            !lookup_pending_r && wb_walk_state_r == WB_IDLE && cache_available;
442        1/1                for (int unsigned slot = 0; slot < WRITE_MISS_ENTRIES; slot++) begin
443        1/1                  if (!rsp_wm_match_found && wm_state_r[slot] == WM_JOINED &&
444                                 wm_word_addr_r[slot] == mshr_word_addr_r[rsp_mshr_slot]) begin
445        0/1     ==>            rsp_wm_match_found = 1'b1;
446        0/1     ==>            rsp_wm_match_slot = slot[WM_W-1:0];
447                             end
                        MISSING_ELSE
448                           end
449                         end
                        MISSING_ELSE
450        1/1              if (rsp_is_write_miss)
451        0/1     ==>        wm_rsp_ready = wm_state_r[rsp_wm_slot] == WM_WAITING &&
                        MISSING_ELSE
452                                          !lookup_pending_r && wb_walk_state_r == WB_IDLE && cache_available;
453                       end
454                     
455                       assign backing_if.rsp_ready = backing_if.rsp.valid &&
456                           (rsp_is_wb || rsp_is_direct || mshr_rsp_ready || wm_rsp_ready);
457                       assign backing_rsp_fire = backing_if.rsp_ready;
458                     
459                       // Dirty words always win the backing port. Consequently, a refill request
460                       // cannot overtake the writeback of the line it is about to replace.
461                       // The remaining priority is direct tile writes, write-miss reads, then
462                       // ordinary read-MSHR refills. All requests use the same ordered physical
463                       // memory link, so this priority also defines response context order.
464                       always_comb begin
465        1/1              backing_if.req = '0;
466        1/1              if (wb_count_r != 0 && !wb_issued_r) begin
467        0/1     ==>        backing_if.req.valid = 1'b1;
468        0/1     ==>        backing_if.req.write = 1'b1;
469        0/1     ==>        backing_if.req.key = {wb_addr_r[wb_head_r], 2'b00};
470        0/1     ==>        backing_if.req.wdata = wb_data_r[wb_head_r];
471        0/1     ==>        backing_if.req.wmask = 32'hffff_ffff;
472        0/1     ==>        backing_if.req.response_context = RESPONSE_CONTEXT_WRITEBACK;
473        1/1              end else if (direct_state_r == DIRECT_REQUEST) begin
474        0/1     ==>        backing_if.req.valid = 1'b1;
475        0/1     ==>        backing_if.req.write = 1'b1;
476        0/1     ==>        backing_if.req.key = {direct_req_r.addr, 2'b00};
477        0/1     ==>        backing_if.req.wdata = direct_req_r.wdata;
478        0/1     ==>        backing_if.req.wmask = direct_req_r.wmask;
479        0/1     ==>        backing_if.req.response_context = RESPONSE_CONTEXT_TILE_WRITE;
480        1/1              end else if (wb_count_r == 0 && !wb_issued_r && wm_issue_found) begin
481        0/1     ==>        backing_if.req.valid = 1'b1;
482        0/1     ==>        backing_if.req.key = {wm_word_addr_r[wm_issue_slot], 2'b00};
483        0/1     ==>        backing_if.req.response_context = RESPONSE_CONTEXT_WRITE_MISS_BASE + wm_issue_slot;
484        1/1              end else if (wb_count_r == 0 && !wb_issued_r && issue_found) begin
485        0/1     ==>        backing_if.req.valid = 1'b1;
486        0/1     ==>        backing_if.req.key = {mshr_word_addr_r[issue_slot], 2'b00};
487        0/1     ==>        backing_if.req.response_context =
488                               {{(RESPONSE_CONTEXT_W-MSHR_W){1'b0}}, issue_slot};
489                         end
                        MISSING_ELSE
490                       end
491                       assign backing_req_fire = backing_if.req.valid && backing_if.req_ready;
492                     
493                       // A replacement is queued only when the old set is valid and dirty. Clean
494                       // victims can simply be overwritten; dirty victims must reach the FPGA.
495                       assign enqueue_runtime = lookup_can_finish && !lookup_hit && cache_lookup_valid &&
496                                                cache_lookup_dirty;
497                       assign enqueue_walk = wb_walk_state_r == WB_CHECK && cache_lookup_valid &&
498                                             cache_lookup_dirty && wb_count_r < WRITEBACK_DEPTH;
499                       assign enqueue_wb = enqueue_runtime || enqueue_walk;
500                       assign enqueue_set = enqueue_runtime ? cache_set(lookup_req_r.addr) : wb_walk_set_r;
501                       assign enqueue_addr = cache_word_addr(cache_lookup_tag, enqueue_set);
502                       assign dequeue_wb = backing_rsp_fire && rsp_is_wb;
503                     
504                       // One synchronous cache port serves request lookups, refills, and the
505                       // boundary walker. Admission and response-ready prevent simultaneous users.
506                       // Refill writes have higher priority than new lookup requests because the
507                       // returned word must become resident before any dependent request proceeds.
508                       always_comb begin
509        1/1              cache_lookup = 1'b0;
510        1/1              cache_lookup_set = '0;
511        1/1              cache_write = 1'b0;
512        1/1              cache_write_set = '0;
513        1/1              cache_write_valid = 1'b1;
514        1/1              cache_write_dirty = 1'b0;
515        1/1              cache_write_tag = '0;
516        1/1              cache_write_data = '0;
517                     
518        1/1              if (lookup_req_r.write && lookup_can_finish && lookup_hit) begin
519        0/1     ==>        cache_write = 1'b1;
520        0/1     ==>        cache_write_set = cache_set(lookup_req_r.addr);
521        0/1     ==>        cache_write_tag = cache_tag(lookup_req_r.addr);
522        0/1     ==>        cache_write_data = lookup_write_value;
523        0/1     ==>        cache_write_dirty = 1'b1;
524        1/1              end else if (backing_rsp_fire && rsp_is_mshr &&
525                                      address_region(mshr_word_addr_r[rsp_mshr_slot]) != REGION_INFER &&
526                                      HAS_CACHE && control_if.control.cfg.cache_enable) begin
527        0/1     ==>        cache_write = 1'b1;
528        0/1     ==>        cache_write_set = cache_set(mshr_word_addr_r[rsp_mshr_slot]);
529        0/1     ==>        cache_write_tag = cache_tag(mshr_word_addr_r[rsp_mshr_slot]);
530        0/1     ==>        cache_write_data = rsp_wm_match_found ?
531                               masked_merge(backing_if.rsp.rdata, wm_data_r[rsp_wm_match_slot],
532                                            wm_mask_r[rsp_wm_match_slot]) : backing_if.rsp.rdata;
533        0/1     ==>        cache_write_dirty = rsp_wm_match_found;
534        1/1              end else if (backing_rsp_fire && rsp_is_write_miss && HAS_CACHE && control_if.control.cfg.cache_enable) begin
535        0/1     ==>        cache_write = 1'b1;
536        0/1     ==>        cache_write_set = cache_set(wm_word_addr_r[rsp_wm_slot]);
537        0/1     ==>        cache_write_tag = cache_tag(wm_word_addr_r[rsp_wm_slot]);
538        0/1     ==>        cache_write_data = masked_merge(backing_if.rsp.rdata, wm_data_r[rsp_wm_slot],
539                                                           wm_mask_r[rsp_wm_slot]);
540        0/1     ==>        cache_write_dirty = 1'b1;
541        1/1              end else if (wb_walk_state_r == WB_READ) begin
542        0/1     ==>        cache_lookup = 1'b1;
543        0/1     ==>        cache_lookup_set = wb_walk_set_r;
544        1/1              end else if (request_accept && request_lookup && !request_merge) begin
545        0/1     ==>        cache_lookup = 1'b1;
546        0/1     ==>        cache_lookup_set = cache_set(vs_if.req.addr);
547                         end
                        MISSING_ELSE
548                       end

550                       mem_io_cache #(
551                         .SETS (CACHE_SETS)
552                       ) i_vs_cache_bank (
553                         .clk_i, .rst_ni, .dft_shift_i,
554                         .invalidate_i (control_if.control.cache_invalidate),
555                         .available_o (cache_available), .invalidated_o (cache_invalidated),
556                         .lookup_i (cache_lookup), .lookup_set_i (cache_lookup_set),
557                         .lookup_valid_o (cache_lookup_valid), .lookup_dirty_o (cache_lookup_dirty),
558                         .lookup_tag_o (cache_lookup_tag), .lookup_data_o (cache_lookup_data),
559                         .dirty_any_o (cache_dirty_any), .dirty_set_o (cache_dirty_set),
560                         .clear_dirty_i (enqueue_walk), .clear_dirty_set_i (wb_walk_set_r),
561                         .write_i (cache_write), .write_set_i (cache_write_set),
562                         .write_valid_i (cache_write_valid), .write_dirty_i (cache_write_dirty),
563                         .write_tag_i (cache_write_tag), .write_data_i (cache_write_data)
564                       );

566                       // The arbiter sees a one-cycle response and forwards it to the tile indicated
567                       // by response_r.id. response_r.valid is cleared on the next clock because
568                       // the tile-facing protocol has no response-ready handshake.
569                       assign vs_if.rsp = response_r;
570                     
571                       always_comb begin
572                         logic mshr_idle;                        // Reduction result: all MSHRs are inactive.
573                         logic write_miss_idle;                  // Reduction result: no write refill remains.
574        1/1              mshr_idle = 1'b1;
575        1/1              for (int unsigned slot = 0; slot < MSHRS; slot++)
576        1/1                mshr_idle &= mshr_state_r[slot] == MSHR_INACTIVE;
577        1/1              write_miss_idle = 1'b1;
578        1/1              for (int unsigned slot = 0; slot < WRITE_MISS_ENTRIES; slot++)
579        1/1                write_miss_idle &= wm_state_r[slot] == WM_INACTIVE;
580        1/1              control_if.vs_status.idle = mshr_idle && write_miss_idle && !lookup_pending_r &&
581                                         direct_state_r == DIRECT_IDLE &&
582                                         wb_count_r == 0 && !wb_issued_r && !response_r.valid &&
583                                         wb_walk_state_r == WB_IDLE && cache_available;
584        1/1              control_if.vs_status.invalidated = !HAS_CACHE || cache_invalidated;
585        1/1              control_if.vs_status.writeback_done = !HAS_CACHE ||
586                           !control_if.control.cfg.cache_enable || wb_walk_done_r;
587                       end
588                     
589                       // All MSHR, lookup, direct-write, response, and writeback state changes are
590                       // committed on this edge. Combinational blocks above only propose actions.
591                       always_ff @(posedge clk_i or negedge rst_ni) begin
592        1/1              if (!rst_ni) begin
593        1/1                for (int unsigned slot = 0; slot < MSHRS; slot++) begin
594        1/1                  mshr_state_r[slot] <= MSHR_INACTIVE;
595        1/1                  mshr_data_r[slot] <= '0;
596        1/1                  mshr_word_addr_r[slot] <= '0;
597        1/1                  mshr_clients_r[slot] <= '0;
598                           end
599        1/1                for (int unsigned slot = 0; slot < WRITE_MISS_ENTRIES; slot++) begin
600        1/1                  wm_state_r[slot] <= WM_INACTIVE;
601        1/1                  wm_word_addr_r[slot] <= '0;
602        1/1                  wm_data_r[slot] <= '0;
603        1/1                  wm_mask_r[slot] <= '0;
604                           end
605        1/1                lookup_pending_r <= 1'b0;
606        1/1                lookup_req_r <= '0;
607        1/1                direct_state_r <= DIRECT_IDLE;
608        1/1                direct_req_r <= '0;
609        1/1                wb_head_r <= '0;
610        1/1                wb_tail_r <= '0;
611        1/1                wb_count_r <= '0;
612        1/1                wb_issued_r <= 1'b0;
613        1/1                wb_walk_state_r <= WB_IDLE;
614        1/1                wb_walk_set_r <= '0;
615        1/1                wb_walk_done_r <= 1'b0;
616        1/1                response_r <= '0;
617        1/1                issue_rr_r <= '0;
618        1/1                wm_issue_rr_r <= '0;
619                         end else begin
620        1/1                response_r.valid <= 1'b0;
621                     
622                           // New requests merge with pending work, start one cache lookup, or enter
623                           // the narrow direct-write slot used by uncached regions.
624        1/1                if (request_accept) begin
625        0/1     ==>          if (request_merge) begin
626        0/1     ==>            mshr_clients_r[match_slot] <= mshr_clients_r[match_slot] | vs_if.req.id;
627        0/1     ==>          end else if (request_merge_write) begin
628        0/1     ==>            wm_data_r[wm_match_slot] <= masked_merge(wm_data_r[wm_match_slot],
629                                                                        vs_if.req.wdata,
630                                                                        vs_if.req.wmask);
631        0/1     ==>            wm_mask_r[wm_match_slot] <= wm_mask_r[wm_match_slot] | vs_if.req.wmask;
632        0/1     ==>          end else if (request_join_read) begin
633        0/1     ==>            wm_state_r[wm_free_slot] <= WM_JOINED;
634        0/1     ==>            wm_word_addr_r[wm_free_slot] <= vs_if.req.addr;
635        0/1     ==>            wm_data_r[wm_free_slot] <= vs_if.req.wdata;
